cmake_minimum_required(VERSION 3.25)
project(PicLens VERSION 4.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
include(CTest)
include(GNUInstallDirs)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Svg Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WEBP REQUIRED IMPORTED_TARGET libwebp)
qt_standard_project_setup(REQUIRES 6.8)
add_library(piclens-domain STATIC src/domain.cpp src/fileoperations.cpp)
target_include_directories(piclens-domain PUBLIC src)
target_link_libraries(piclens-domain PUBLIC Qt6::Core Qt6::Gui)
add_library(piclens-imaging STATIC src/imaging.cpp src/imageitem.cpp)
target_include_directories(piclens-imaging PUBLIC src)
target_link_libraries(piclens-imaging PUBLIC Qt6::Core Qt6::Gui Qt6::Quick Qt6::Concurrent)
qt_add_executable(piclens-worker src/worker_main.cpp)
target_link_libraries(piclens-worker PRIVATE Qt6::Core Qt6::Gui PkgConfig::WEBP)
qt_add_executable(piclens src/main.cpp src/controller.cpp src/controller.h src/services.cpp src/services.h src/models.cpp src/models.h)
target_link_libraries(piclens PRIVATE piclens-domain piclens-imaging Qt6::QuickControls2 Qt6::Svg)
target_compile_definitions(piclens PRIVATE PICLENS_VERSION="${PROJECT_VERSION}")
file(GLOB QML_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" qml/*.qml)
qt_add_resources(piclens qml PREFIX "/" FILES ${QML_FILES})
file(GLOB ICON_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/Icons/Lucide/*.svg")
qt_add_resources(piclens icons PREFIX "/icons" BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/Icons/Lucide" FILES ${ICON_FILES})
qt_add_resources(piclens brand PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../assets" FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/Square150x150Logo.scale-200.png")
add_dependencies(piclens piclens-worker)
install(TARGETS piclens RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS piclens-worker RUNTIME DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/piclens)
if(UNIX AND NOT APPLE)
    install(FILES ../../packaging/arch/piclens.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES ../../packaging/arch/piclens.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/48x48/apps)
    install(FILES ../../packaging/arch/io.github.Mingx94.PicLens.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
    install(FILES ../../LICENSE THIRD-PARTY.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/piclens)
    install(FILES ../../assets/Icons/Lucide/LICENSE.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/piclens RENAME Lucide-LICENSE.txt)
endif()
if(BUILD_TESTING)
    foreach(suite domain imaging app controller)
        add_executable(${suite}_test tests/${suite}_test.cpp)
        target_compile_definitions(${suite}_test PRIVATE PICLENS_REPO="${CMAKE_CURRENT_SOURCE_DIR}/../.." PICLENS_WORKER="$<TARGET_FILE:piclens-worker>")
        target_link_libraries(${suite}_test PRIVATE piclens-domain piclens-imaging Qt6::Test)
        add_dependencies(${suite}_test piclens-worker)
        add_test(NAME ${suite} COMMAND ${suite}_test)
        # Desktop themes such as gtk3 can require a display even with offscreen.
        set_tests_properties(${suite} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=" TIMEOUT 120)
    endforeach()
    target_sources(app_test PRIVATE src/services.cpp src/models.cpp)
    target_sources(controller_test PRIVATE src/controller.cpp src/services.cpp src/models.cpp)

    add_executable(qml_test tests/qml_test.cpp src/controller.cpp src/services.cpp src/models.cpp)
    target_compile_definitions(qml_test PRIVATE PICLENS_REPO="${CMAKE_CURRENT_SOURCE_DIR}/../.." PICLENS_WORKER="$<TARGET_FILE:piclens-worker>")
    target_link_libraries(qml_test PRIVATE piclens-domain piclens-imaging Qt6::QuickControls2 Qt6::Svg Qt6::Test)
    add_dependencies(qml_test piclens-worker)
    qt_add_resources(qml_test qml PREFIX "/" FILES ${QML_FILES})
    qt_add_resources(qml_test icons PREFIX "/icons" BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../assets/Icons/Lucide" FILES ${ICON_FILES})
    add_test(NAME qml COMMAND qml_test)
    set_tests_properties(qml PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=" TIMEOUT 180)
    add_executable(metrics_test tests/metrics_test.cpp)
    target_compile_definitions(metrics_test PRIVATE PICLENS_REPO="${CMAKE_CURRENT_SOURCE_DIR}/../.." PICLENS_WORKER="$<TARGET_FILE:piclens-worker>" PICLENS_APP="$<TARGET_FILE:piclens>")
    target_link_libraries(metrics_test PRIVATE piclens-domain piclens-imaging Qt6::Test)
    add_dependencies(metrics_test piclens-worker piclens)
    target_sources(metrics_test PRIVATE src/controller.cpp src/services.cpp src/models.cpp)
    add_test(NAME metrics COMMAND metrics_test)
    set_tests_properties(metrics PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=" TIMEOUT 120)
endif()
