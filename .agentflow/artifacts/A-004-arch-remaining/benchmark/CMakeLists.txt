cmake_minimum_required(VERSION 3.25)
project(corpus_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Quick QuickControls2 Concurrent Svg Test)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(SOURCE "${REPO}/apps/linux")
set(PICLENS_BUILD "/tmp/piclens-arch-validation.IHj5Y4Ng/build" CACHE PATH "Release app build containing static domain/imaging libraries")
add_executable(corpus_check main.cpp ${SOURCE}/src/controller.cpp ${SOURCE}/src/services.cpp ${SOURCE}/src/models.cpp)
target_include_directories(corpus_check PRIVATE ${SOURCE}/src)
target_link_libraries(corpus_check PRIVATE ${PICLENS_BUILD}/libpiclens-domain.a ${PICLENS_BUILD}/libpiclens-imaging.a Qt6::Widgets Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg Qt6::Test)
file(GLOB QML_FILES ${SOURCE}/qml/*.qml)
qt_add_resources(corpus_check qml PREFIX /qml BASE ${SOURCE}/qml FILES ${QML_FILES})
file(GLOB ICON_FILES ${REPO}/assets/Icons/Lucide/*.svg)
qt_add_resources(corpus_check icons PREFIX /icons BASE ${REPO}/assets/Icons/Lucide FILES ${ICON_FILES})
