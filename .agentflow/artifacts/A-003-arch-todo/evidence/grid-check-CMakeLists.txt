cmake_minimum_required(VERSION 3.25)
project(grid_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Concurrent Svg Test)
set(SOURCE /home/michael/Work/PicLens/apps/linux)
add_executable(grid_check main.cpp ${SOURCE}/src/controller.cpp ${SOURCE}/src/services.cpp ${SOURCE}/src/models.cpp)
target_include_directories(grid_check PRIVATE ${SOURCE}/src)
target_link_libraries(grid_check PRIVATE /tmp/piclens-arch-validation.IHj5Y4Ng/build/libpiclens-domain.a /tmp/piclens-arch-validation.IHj5Y4Ng/build/libpiclens-imaging.a Qt6::QuickControls2 Qt6::Concurrent Qt6::Svg Qt6::Test)
file(GLOB QML_FILES ${SOURCE}/qml/*.qml)
qt_add_resources(grid_check qml PREFIX /qml BASE ${SOURCE}/qml FILES ${QML_FILES})
file(GLOB ICON_FILES /home/michael/Work/PicLens/assets/Icons/Lucide/*.svg)
qt_add_resources(grid_check icons PREFIX /icons BASE /home/michael/Work/PicLens/assets/Icons/Lucide FILES ${ICON_FILES})
